\section*{Introduction}
This report computes the required hydraulic pressure for a given motor displacement and operating condition. It shows step-by-step traction, torque and pressure calculations.

\section{Theory and Formulas}
\begin{itemize}
  \item Torque requirement: $T = F \times r$
  \item Convert torque to motor shaft (gearbox input) if required: $T_{motor} = T_{axle} / gearbox\_ratio$
  \item Torque (kg·cm): $T_{kg\cdot cm} = T_{Nm} \times 10.1972$
  \item Pressure from motor displacement: $P = \dfrac{T_{kg\cdot cm} \times 2\pi}{D_{motor} \times \eta_m}$
\end{itemize}

\section{Step-by-step Calculation}
\subsection*{Step 1: Speed and Wheel RPM}
\[ V = {{ speed | default(0) }} \times \frac{1000}{3600} = {{ speed_mps | default(0) | round(2) }} \, m/s \]
\[ C = \frac{ {{ wheel_diameter | default(0) }} \times \pi }{1000} = {{ wheel_circumference | default(0) | round(3) }} \, m \]
\[ RPM = \frac{ {{ speed_mps | default(0) | round(2) }} }{ {{ wheel_circumference | default(0) | round(3) }} } \times 60 = {{ wheel_rpm | default(0) | round(2) }} \, RPM \]

\subsection*{Resistance & Torque}
\[ F_{total} = {{ total_resistance | default(0) | round(2) }} \, kN \]
\[ r = \frac{ {{ wheel_diameter | default(0) }} }{2000} = {{ wheel_radius | default(0) | round(3) }} \; m \]
\[ T_{total} = F_{total} \times r = {{ required_total_torque | default(0) | round(2) }} \; Nm \]
\[ T_{per\_axle} = {{ per_axle_torque | default(0) | round(2) }} \; Nm \]
\[ T_{per\_motor} = {{ per_motor_torque | default(0) | round(2) }} \; Nm \]
\[ T_{motor\_shaft} (gearbox input) = {{ per_gearbox_input_torque | default(0) | round(2) }} \; Nm \]

\subsection*{Pressure from Motor Displacement}
Convert torque to kg·cm:
\[ T_{kg\,cm} = {{ per_gearbox_input_torque_kg_cm | default(0) | round(2) }} \; kg\cdot cm \]
Using \( P = \frac{T_{kg\,cm} \times 2\pi}{D \times \eta_m} \):
\[ Required\ Pressure = \frac{ {{ per_gearbox_input_torque_kg_cm | default(0) | round(2) }} \times 2\pi }{ {{ motor_displacement_cc | default(0) | round(2) }} \times {{ (mech_eff_motor | default(100) / 100) | round(4) }} } = {{ required_pressure_bar | default(0) | round(2) }} \; bar \]
\[ = {{ pressure_kg_cm2 | default(0) | round(2) }} \; \text{kg/cm}^2 \]
\subsection*{Detailed worked calculation (formula → substitution → numeric result)}
\begin{itemize}
  \item Torque (per motor, gearbox input): $T = {{ per_gearbox_input_torque | default(0) | round(2) }}$ Nm = ${{ per_gearbox_input_torque_kg_cm | default(0) | round(2) }}$ kg·cm.
  \item Motor displacement used: $D = {{ motor_displacement_cc | round(2) }}$ cc/rev.
  \item Mechanical efficiency used: $\eta_m = {{ mech_eff_motor | default(0) }}\%$.
  \item Substitution into pressure formula gives: $P = {{ required_pressure_bar | round(2) }}$ bar ({{ pressure_kg_cm2 | round(2) }} kg/cm$^2$).
\end{itemize}

\subsection*{Units & assumptions}
\begin{itemize}
  \item Pressure unit conversion: 1 bar = 1.01972 kg/cm$^2$.
  \item Torque conversion: 1 Nm = 10.1972 kg·cm.
  \item Assumptions: motor displacement is exact input; mechanical efficiencies represent shaft losses only.
\end{itemize}
